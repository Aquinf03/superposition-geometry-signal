SDK

Import from spg. Package source lives under sdk/spg/.

Tracker / SignalLogger

Append-only logger. Writes signals.csv + meta.json under results_dir/run_name/. With live_print=True, prints a live line each log.

python
from spg import Tracker

t = Tracker(results_dir="runs", run_name="demo", live_print=True)
t.log(step=0, loss=1.2, geometry={
    "interference_mean": 0.4,
    "spectral_participation": 0.2,
    "coactivation_overlap": 0.1,
})
# multi-layer keys also work: geometry_L8_interference_mean or L8_interference_mean

Diff

Snapshot a query against a neighbor bank, then compare two snapshots. Summary score is lower when neighborhoods are closer.

python
from spg import Diff

a = Diff.snapshot(query_a, bank, feature_id="eiffel", step=19)
b = Diff.snapshot(query_b, bank, feature_id="louvre", step=39)
d = Diff.compare(a, b)
print(d["summary_score"])

plot · write_view_html

plot(csv) writes a loss + geometry PNG. write_view_html(run_dir) builds an interactive neighborhood scrubber.

python
from spg import plot, write_view_html
from pathlib import Path

plot("runs/demo/signals.csv")
write_view_html(Path("experiments/results/train_gpt2_small_geometry"))

GeoControl

Soft regularizer. Off by default. Point target or band; add penalty_on_value(metric_t) to the LM loss after warmup. See Control.

python
from spg import GeoControl, format_control_line

control = GeoControl(
    layer=8, metric="interference_mean",
    target=0.35, weight=1e-2, enabled=True,
)
# loss = lm_loss + control.penalty_on_value(metric_t)
print(format_control_line(step=t, loss=float(lm_loss),
                          geometry=geometry, control=control.status(geometry)))
